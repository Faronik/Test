cmake_minimum_required(VERSION 3.20)
project(toolstock VERSION 2.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PostgreSQL REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
if(WIN32)
  find_package(httplib CONFIG REQUIRED)
  set(TOOLSTOCK_HTTP httplib::httplib)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(HTTPLIB REQUIRED IMPORTED_TARGET cpp-httplib)
  set(TOOLSTOCK_HTTP PkgConfig::HTTPLIB)
endif()
add_executable(toolstock src/main.cpp src/import.cpp src/runtime.cpp)
target_include_directories(toolstock PRIVATE ${PostgreSQL_INCLUDE_DIRS})
target_link_libraries(toolstock PRIVATE ${TOOLSTOCK_HTTP} PostgreSQL::PostgreSQL OpenSSL::SSL OpenSSL::Crypto SQLite::SQLite3 Threads::Threads nlohmann_json::nlohmann_json)
if(MSVC)
  target_compile_options(toolstock PRIVATE /W4 /utf-8 /EHsc)
  target_compile_definitions(toolstock PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(toolstock PRIVATE advapi32 ws2_32 crypt32)
else()
  target_compile_options(toolstock PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS toolstock RUNTIME DESTINATION .)
install(DIRECTORY web sql docs DESTINATION .)
install(FILES README.md DESTINATION .)
